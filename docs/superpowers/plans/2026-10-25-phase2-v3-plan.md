```python
"""Token helpers shared by the Evidence agent, Critic and deletion test.
Tokenisation matches sklearn's default token_pattern on lowercased text, so phrases
from the TF-IDF vocabulary can be checked against any text."""
import re

TOKEN = re.compile(r"(?u)\b\w\w+\b")
QUOTE = re.compile(r'["“”]([^"“”]+)["“”]')


def tokens(s: str) -> list[str]:
    return TOKEN.findall(s.lower())


def contains(hay: list[str], phrase: str) -> bool:
    p = tokens(phrase); n = len(p)
    return n > 0 and any(hay[i:i + n] == p for i in range(len(hay) - n + 1))


def display_span(text: str, phrase: str) -> str:
    """Original-text span for a vocabulary phrase ("doesn work" -> "doesn't work"); the phrase itself if absent."""
    p = tokens(phrase)
    if not p:
        return phrase
    m = re.search(r"(?<!\w)" + r"\W+(?:\w\W+)?".join(map(re.escape, p)) + r"(?!\w)", text, re.I)
    return m.group(0) if m else phrase


def quoted(s: str) -> list[str]:
    return QUOTE.findall(s)


def token_spans(text: str) -> list[tuple[int, int]]:
    return [m.span() for m in TOKEN.finditer(text)]


def phrase_token_idx(text: str, phrase: str) -> list[int] | None:
    """Indices (into token_spans(text)) of the first contiguous occurrence of phrase, or None."""
    hay, p = tokens(text), tokens(phrase); n = len(p)
    for i in range(len(hay) - n + 1):
        if n and hay[i:i + n] == p:
            return list(range(i, i + n))
    return None


def drop_tokens(text: str, idx: set[int]) -> str:
    """Remove the chosen word tokens, keep every other character (same mechanism for cited and random deletion)."""
    out, last = [], 0
    for i, (a, b) in enumerate(token_spans(text)):
        if i in idx:
            out.append(text[last:a]); last = b
    out.append(text[last:])
    return "".join(out)
```

```python
"""Evidence agent: E6b TF-IDF + LR. Probabilities plus per-phrase contributions (coef[star] x tf-idf weight)."""
from dataclasses import dataclass, field
import numpy as np
from agentic_sentiment.v3.text import display_span


def evidence_text(title: str, text: str) -> str:
    return (title + " . " if title else "") + text          # same as evid_H.py txt() with title=True


@dataclass
class EvidenceAgent:
    vectorizer: object
    model: object
    names: np.ndarray = field(init=False, repr=False)

    def __post_init__(self):
        self.names = self.vectorizer.get_feature_names_out()

    def probs(self, title: str, text: str) -> np.ndarray:
        return self.model.predict_proba(self.vectorizer.transform([evidence_text(title, text)]))[0]

    def phrases(self, title: str, text: str, star: int, k_for: int = 5, k_against: int = 3):
        full = evidence_text(title, text)
        x = self.vectorizer.transform([full]).tocsr()
        k = list(self.model.classes_).index(star)
        contrib = x.data * self.model.coef_[k, x.indices]
        order = np.argsort(contrib)
        fmt = lambda j: {"phrase": str(self.names[x.indices[j]]), "display": display_span(full, str(self.names[x.indices[j]])),
                         "weight": round(float(contrib[j]), 4)}
        pos = [fmt(j) for j in order[::-1] if contrib[j] > 0][:k_for]
        neg = [fmt(j) for j in order if contrib[j] < 0][:k_against]
        return pos, neg
```

```python
"""Reflector (LR on agents' log-probabilities, as SH) with an Evidence-only fallback, and the Router threshold search."""
from dataclasses import dataclass
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.model_selection import StratifiedKFold

EPS = 1e-6
GRID = np.round(np.arange(0.30, 0.96, 0.05), 2)


def logp(p) -> np.ndarray:
    return np.log(np.clip(np.asarray(p, float), EPS, 1))


@dataclass
class Reflector:
    full: LogisticRegression       # features [log probe(5), log evidence(5)], labels 0-4
    fallback: LogisticRegression   # features [log evidence(5)]

    def combine(self, probe_p, evid_p):
        if probe_p is None:
            P = self.fallback.predict_proba(logp(evid_p)[None])[0]; shares = {"probe": 0.0, "evidence": 1.0}
        else:
            x = np.r_[logp(probe_p), logp(evid_p)][None]; P = self.full.predict_proba(x)[0]
            c = self.full.coef_[int(P.argmax())] * x[0]; a, b = abs(c[:5].sum()), abs(c[5:].sum()); s = (a + b) or 1.0
            shares = {"probe": a / s, "evidence": b / s}
        return int(P.argmax()) + 1, P, shares


def fit_reflector(P, E, stars) -> Reflector:
    y = np.asarray(stars) - 1
    return Reflector(LogisticRegression(C=1, max_iter=3000).fit(np.c_[logp(P), logp(E)], y),
                     LogisticRegression(C=1, max_iter=3000).fit(logp(E), y))


def choose_tau(P, E, stars, grid=None, tol=0.005, seed=0):
    """Smallest tau whose routed 5-fold CV accuracy is within tol of full SH (skip the probe when max Evidence prob >= tau)."""
    grid = GRID if grid is None else grid; y = np.asarray(stars)
    full_ok = np.zeros(len(y), bool); routed = {t: np.zeros(len(y), bool) for t in grid}
    for tr, te in StratifiedKFold(5, shuffle=True, random_state=seed).split(E, y):
        r = fit_reflector(P[tr], E[tr], y[tr])
        fp = r.full.predict(np.c_[logp(P[te]), logp(E[te])]) + 1; bp = r.fallback.predict(logp(E[te])) + 1
        full_ok[te] = fp == y[te]; conf = E[te].max(1)
        for t in grid:
            routed[t][te] = np.where(conf >= t, bp, fp) == y[te]
    base = float(full_ok.mean())
    rows = [(float(t), float(routed[t].mean()), float((E.max(1) >= t).mean())) for t in grid]
    ok = [t for t, acc, _ in rows if acc >= base - tol]
    return (min(ok) if ok else None), base, rows
```

```python
"""One LLaMA-3-8B Q4 model in memory for both phases: Phase 1 = adapter R on (first-digit probabilities, as eval_p1.py);
Phase 2 probe = LoRA scale 0 (base model, L20 last-token state, as probe_agent4.py)."""
import math
import joblib
import numpy as np
from agentic_sentiment.v3 import paths as PT

PROBE_PROMPT = "Review: {text}\nRating (1-5):"


class SharedLlama:
    def __init__(self, base: str = PT.MLX_BASE, adapter: str | None = PT.ADAPTER_R):
        from mlx_lm import load
        from mlx_lm.tuner.lora import LoRALinear
        self.model, self.tok = load(base, adapter_path=adapter)
        self.loras = [m for _, m in self.model.named_modules() if isinstance(m, LoRALinear)]
        self.scale_on = self.loras[0].scale if self.loras else 0.0
        self.sp = self.tok.encode(" ", add_special_tokens=False)

    def lora(self, on: bool):
        for m in self.loras:
            m.scale = self.scale_on if on else 0.0


class ProbeAgent:
    def __init__(self, llama: SharedLlama, scaler, clf):
        self.llama, self.scaler, self.clf = llama, scaler, clf

    def hidden(self, text: str) -> np.ndarray:
        import mlx.core as mx
        from mlx_lm.models.base import create_attention_mask
        self.llama.lora(False); m = self.llama.model
        ids = self.llama.tok.encode(PROBE_PROMPT.format(text=text[:1200]), add_special_tokens=False) + self.llama.sp
        h = m.model.embed_tokens(mx.array(ids)[None]); mask = create_attention_mask(h, None)
        for layer in m.model.layers[:20]:
            h = layer(h, mask, None)
        v = h[0, -1]; mx.eval(v)
        return np.array(v.astype(mx.float16)).astype(np.float32)

    def probs(self, text: str) -> np.ndarray:
        return self.clf.predict_proba(self.scaler.transform(self.hidden(text)[None]))[0]


def load_probe(llama: SharedLlama) -> ProbeAgent:
    d = joblib.load(PT.PROBE_JOBLIB)
    return ProbeAgent(llama, d["scaler"], d["clf"])


def phase1_probs(llama: SharedLlama, text: str) -> np.ndarray:
    """Phase 1 Analyst (LLaMA + LoRA arm R): softmax over first-digit logits 1-5, same prompt as eval_p1.py."""
    import mlx.core as mx
    from agentic_sentiment.phase1.config import DataConfig
    from agentic_sentiment.phase1.data.preprocessing import build_prompt
    llama.lora(True); tok = llama.tok
    dig = [tok.encode(str(d), add_special_tokens=False)[-1] for d in range(1, 6)]
    ids = tok.encode(build_prompt(text[:1200], DataConfig(use_one_shot=False)), add_special_tokens=False) + llama.sp
    lg = llama.model(mx.array(ids)[None])[0, -1]; v = [float(lg[d]) for d in dig]; mx_ = max(v); e = [math.exp(x - mx_) for x in v]
    return np.array(e) / sum(e)
```

```python
    return fp, np.where(skip, bp, fp), skip, bp
```

```python
"""Critic for explanations: 4 rule checks + 1 LLM check, each failure turned into specific feedback for the Explainer."""
import re
from agentic_sentiment.v3.text import tokens, contains, quoted

STAR = re.compile(r"\b([1-5])(?:\s*|-)(?:stars?|★)", re.I)
DISAGREE = ("disagree", "differ", "split", "conflict")
JUDGE_SYSTEM = ("You check an explanation against a product review. Reply exactly NO if every claim in the explanation is supported by "
                'the review. Otherwise reply YES: "<the unsupported words>".')


def judge_user(rationale: str, title: str, text: str) -> str:
    return f"Review title: {title}\nReview: {text[:1200]}\n\nExplanation: {rationale}\n\nDoes the explanation claim anything the review does not say?"


def parse_judge(answer: str):
    a = answer.strip()
    if a.upper().startswith("NO"):
        return True, None
    if a.upper().startswith("YES"):
        q = quoted(a)
        return False, (q[0] if q else a[3:].strip(" :"))
    return None, None


def rule_checks(rationale: str, pack: dict, title: str, text: str) -> list[str]:
    fb, s, rt = [], pack["star"], tokens(rationale)
    stars = STAR.findall(rationale)                                                       # 1: stated star = final star
    if not stars:
        fb.append(f"State the final rating as '{s} stars'.")
    for x in sorted({int(x) for x in stars} - {s}):
        fb.append(f"You said {x} stars; the final rating is {s} stars.")
    review, phrases = tokens((title + " . " if title else "") + text), [p["phrase"] for p in pack["for"] + pack["against"]]
    for q in quoted(rationale):                                                           # 2: quotes occur in review and pack
        if not contains(review, q):
            fb.append(f"'{q}' does not occur in the review.")
        elif not any(contains(tokens(q), p) or contains(tokens(p), q) for p in phrases):
            fb.append(f"'{q}' is not one of the evidence phrases.")
    top = pack["for"][:3]; need = min(2, len(top))                                        # 3: cover top phrases
    if sum(contains(rt, p["phrase"]) for p in top) < need:
        fb.append(f"Quote at least {need} of these phrases: " + ", ".join(f'"{p["display"]}"' for p in top))
    if pack["agreed"] is False and not any(w in rationale.lower() for w in DISAGREE):    # 4: disagreement mentioned
        fb.append("The agents disagreed; say so.")
    return fb


def _check_no(f: str) -> int:
    if f.startswith("Quote"): return 3
    if f.endswith("does not occur in the review.") or f.endswith("is not one of the evidence phrases."): return 2
    if f == "The agents disagreed; say so.": return 4
    return 1


def critique(rationale, pack, title, text, judge_fn):
    fb = rule_checks(rationale, pack, title, text); info = {"failed_checks": sorted({_check_no(f) for f in fb}), "judge_unparsed": False}
    if not fb:                                                                            # 5: LLM check, only when the rules pass
        ok, span = parse_judge(judge_fn(JUDGE_SYSTEM, judge_user(rationale, title, text)))
        if ok is None:
            info["judge_unparsed"] = True                                                 # counted and reported; treated as pass
        elif not ok:
            fb.append(f"Remove the unsupported claim: '{span}'."); info["failed_checks"] = [5]
    return fb, info
```

```python
"""SC2 reflective re-rating: on reviews where the main stack is unsure, an instruction-tuned LLM chooses between the stack's top-2 stars
using the review and the evidence phrases for each candidate. Asked twice with the order swapped; inconsistent answers abstain."""
import re

SC_SYSTEM = ("You are a careful reviewer of product-review ratings. Decide which of two star ratings the review deserves. "
             "Reply with the digit only.")


def sc_user(title, text, cand, phr):
    f = lambda ps: ", ".join(f'"{p["display"]}"' for p in ps) or "none"
    a, b = cand
    return (f"Review title: {title}\nReview: {text[:1200]}\n\nCandidate ratings: {a} stars or {b} stars.\n"
            f"Phrases supporting {a} stars: {f(phr[a])}\nPhrases supporting {b} stars: {f(phr[b])}\n\nWhich rating fits the review better: {a} or {b}?")


def parse_choice(answer, cand):
    m = re.search(r"[1-5]", answer)
    return int(m.group(0)) if m and int(m.group(0)) in cand else None


def reflect_pick(chat, title, text, cand, phr):
    first = parse_choice(chat(SC_SYSTEM, sc_user(title, text, cand, phr)), cand)
    second = parse_choice(chat(SC_SYSTEM, sc_user(title, text, cand[::-1], phr)), cand)
    return first if first is not None and first == second else None
```

```python
"""Phase 2 v3 LangGraph: evidence -> router -> [probe] -> reflector -> pack -> explainer -> critic -(rewrite, max 2)-> END.
The specialists decide the star; LLM agents only explain it and check the explanation."""
import time
from typing import TypedDict
import numpy as np
from langgraph.graph import END, StateGraph
from agentic_sentiment.v3.critic import critique
from agentic_sentiment.v3.explainer import EXPLAIN_SYSTEM, explainer_user
from agentic_sentiment.v3.pack import build_pack


class V3State(TypedDict, total=False):
    review_id: str; title: str; review_text: str
    evidence_probs: list; route: str; probe_probs: list | None
    star: int; probs: list; shares: dict; agreed: bool | None; pack: dict
    drafts: list; feedback: list; rounds: int; faithful_first: bool; faithful: bool
    failed_checks: list; judge_unparsed: int; llm_calls: int; timings: dict


def build_graph_v3(evidence, reflector, probe_fn, explain_fn=None, judge_fn=None, tau=None, max_rewrites=2):
    def timed(state, key, t0):
        return {**state.get("timings", {}), key: state.get("timings", {}).get(key, 0.0) + time.time() - t0}

    def evidence_node(s):
        t0 = time.time(); p = evidence.probs(s["title"], s["review_text"])
        return {"evidence_probs": list(map(float, p)), "timings": timed(s, "evidence_s", t0)}

    def router_node(s):
        skip = s.get("probe_probs") is None and tau is not None and max(s["evidence_probs"]) >= tau
        return {"route": "skip" if skip else "probe"}

    def probe_node(s):
        if s.get("probe_probs") is not None:
            return {}
        t0 = time.time(); return {"probe_probs": list(map(float, probe_fn(s["review_text"]))), "timings": timed(s, "probe_s", t0)}

    def reflector_node(s):
        pp = s.get("probe_probs") if s["route"] == "probe" else None
        star, P, shares = reflector.combine(pp, np.array(s["evidence_probs"]))
        agreed = None if pp is None else int(np.argmax(pp)) == int(np.argmax(s["evidence_probs"]))
        return {"star": star, "probs": list(map(float, P)), "shares": shares, "agreed": agreed}

    def pack_node(s):
        pos, neg = evidence.phrases(s["title"], s["review_text"], s["star"])
        return {"pack": build_pack(s["star"], s["probs"], s["shares"], s["agreed"], pos, neg), "drafts": [], "feedback": [], "rounds": 0,
                "failed_checks": [], "judge_unparsed": 0, "llm_calls": 0}

    def explainer_node(s):
        t0 = time.time(); prev = s["drafts"][-1] if s["drafts"] else None
        text = explain_fn(EXPLAIN_SYSTEM, explainer_user(s["pack"], s["title"], s["review_text"], prev, s["feedback"][-1] if prev else ()))
        return {"drafts": s["drafts"] + [text], "rounds": s["rounds"] + (1 if prev else 0), "llm_calls": s["llm_calls"] + 1,
                "timings": timed(s, "explain_s", t0)}

    def critic_node(s):
        t0 = time.time(); fb, info = critique(s["drafts"][-1], s["pack"], s["title"], s["review_text"], judge_fn)
        upd = {"feedback": s["feedback"] + [fb], "failed_checks": s["failed_checks"] + [info["failed_checks"]], "faithful": not fb,
               "judge_unparsed": s["judge_unparsed"] + int(info["judge_unparsed"]),
               "llm_calls": s["llm_calls"] + int(not info["failed_checks"] or info["failed_checks"] == [5] or info["judge_unparsed"]),
               "timings": timed(s, "critic_s", t0)}
        if len(s["drafts"]) == 1:
            upd["faithful_first"] = not fb
        return upd

    g = StateGraph(V3State)
    for n, f in [("evidence", evidence_node), ("router", router_node), ("probe", probe_node), ("reflector", reflector_node)]:
        g.add_node(n, f)
    g.set_entry_point("evidence"); g.add_edge("evidence", "router")
    g.add_conditional_edges("router", lambda s: s["route"], {"probe": "probe", "skip": "reflector"}); g.add_edge("probe", "reflector")
    if explain_fn is None:
        g.add_edge("reflector", END); return g.compile()
    for n, f in [("pack", pack_node), ("explainer", explainer_node), ("critic", critic_node)]:
        g.add_node(n, f)
    g.add_edge("reflector", "pack"); g.add_edge("pack", "explainer"); g.add_edge("explainer", "critic")
    g.add_conditional_edges("critic", lambda s: "explainer" if not s["faithful"] and s["rounds"] < max_rewrites else END,
                            {"explainer": "explainer", END: END})
    return g.compile()
```

Note on `llm_calls` in `critic_node`: the judge is called exactly when the rule checks pass. In that case `failed_checks` is either `[]` (judge said NO, or its answer was unparsable) or `[5]` (judge said YES), and the expression counts 1. In every other case it counts 0. The test `test_rewrite_loop_until_faithful` pins this behaviour.

```python
"""B2 deletion test (declared LOG.md). Per review: SH prob of the explained star on review text only, after removing the cited
phrases, after removing the same number of random words (3 draws), and with only cited / only random words kept. Resumable."""
import json, os, random, sys, time, zlib, joblib, numpy as np
sys.path.insert(0, "src")
from agentic_sentiment.v3 import paths as PT
from agentic_sentiment.v3.evidence import EvidenceAgent

    for b in map(json.loads, open(f"{PT.B2}{split}.jsonl")):

        cited = [q for q in quoted(b["drafts"][-1]) if contains(tokens(t), q)] or [p["display"] for p in b["pack"]["for"][:3]]
        idx = set().union(*[set(phrase_token_idx(t, q) or []) for q in cited]); k = len(idx); nt = len(token_spans(t))
        rng = random.Random(zlib.crc32(b["review_id"].encode())); draws = [set(rng.sample(range(nt), min(k, nt))) for _ in range(3)]
```

```python
"""Demo helpers: live call with cached fallback (never break on stage) and tagged paper numbers for check_demo.py."""
import json, os

CACHE = "presentation/demo_cache.json"


def cached(key, fn, cache_path=CACHE):
    store = json.load(open(cache_path)) if os.path.exists(cache_path) else {}
    if os.environ.get("DEMO_OFFLINE") and key in store:
        print(f"[cached result: {key}]"); return store[key]
    try:
        val = fn()
    except Exception as ex:                       # model missing / too slow: fall back, say so
        if key not in store:
            raise
        print(f"[live run failed ({type(ex).__name__}); showing cached result: {key}]"); return store[key]
    if os.environ.get("DEMO_RECORD"):
        store[key] = val; json.dump(store, open(cache_path, "w"), indent=1)
```
